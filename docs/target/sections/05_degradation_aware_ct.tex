\subsection{Luận đề nghiên cứu và Phạm vi}

\begin{tcolorbox}[colback=blue!3!white,colframe=blue!45!black,arc=2pt,boxrule=0.6pt,top=4pt,bottom=4pt,left=6pt,right=6pt]
\textbf{Luận đề nghiên cứu cốt lõi (Central Research Thesis):}\\[0.2em]
\emph{``$\text{Data Drift} \neq \text{Model Degradation} \neq \text{Retraining Trigger}$. Một hệ thống Continuous Training tin cậy phải xác định chính xác nguyên nhân, mức độ nghiêm trọng (Severity) và độ tin cậy của bằng chứng trước khi quyết định hành động; không tự động tái huấn luyện theo quán tính chỉ dựa trên một cảnh báo drift đơn lẻ.''}
\end{tcolorbox}

\begin{tcolorbox}[colback=orange!4!white,colframe=orange!55!black,arc=2pt,boxrule=0.6pt,top=3pt,bottom=3pt,left=4pt,right=4pt,halign=center]
\small\textbf{Vòng lặp khép kín mục tiêu (Closed-loop Architecture):}\\[0.1em]
Inference + Feedback $\rightarrow$ EvidenceWindow $\rightarrow$ DegradationEvent $\rightarrow$ Multi-signal Diagnosis $\rightarrow$ Risk/Policy Decision $\rightarrow$ Conditional CT / HITL / No-op $\rightarrow$ EvaluationGate $\rightarrow$ Shadow/Canary $\rightarrow$ Promotion / Rollback
\end{tcolorbox}

Hệ thống phân tách rành mạch không gian hành động thành 5 nhánh can thiệp (Action Space):
\begin{enumerate}
  \item \textbf{NO\_OP / OBSERVE (Quan sát):} Ghi nhận sự kiện khi xảy ra trôi dạt lành tính (Benign Drift) hoặc mô hình hoạt động ổn định; tuyệt đối không tái huấn luyện nhằm tiết kiệm chi phí tài nguyên.
  \item \textbf{DATA\_ENGINEERING (Sự cố dữ liệu):} Đình chỉ toàn bộ tiến trình CT khi phát hiện lỗi schema, vi phạm miền giá trị hoặc mất tính tươi mới dữ liệu; chuyển giao cảnh báo cho đội ngũ kỹ thuật (\textbf{Engineering Incident}) để xử lý tận gốc rễ.
  \item \textbf{INCREMENTAL\_RETRAIN (Tái huấn luyện gia tăng):} Kích hoạt cập nhật mô hình trên cửa sổ dữ liệu mới khi trôi dạt có hại ở mức độ vừa phải ($S_2$) và đã có đủ mẫu nhãn kiểm chứng.
  \item \textbf{FULL\_RETRAIN (Tái huấn luyện toàn phần):} Kích hoạt quy trình huấn luyện lại toàn bộ khi suy giảm nghiêm trọng ($S_3, S_4$) và có sự dịch chuyển phân phối sâu rộng.
  \item \textbf{HITL (Human-in-the-Loop):} Chuyển giao chuyên gia ML thẩm định và ra quyết định khi tín hiệu suy giảm mâu thuẫn, thiếu nhãn kiểm chứng hoặc suy giảm chưa rõ nguyên nhân.
\end{enumerate}

\subsection{Câu hỏi nghiên cứu và Giả thuyết khoa học}
Để chứng minh đóng góp học thuật vượt ra ngoài phạm vi xây dựng tính năng kỹ thuật, quá trình thực nghiệm sẽ tập trung kiểm chứng câu hỏi nghiên cứu trung tâm:

\begin{quote}
\emph{``Khi mô hình sản xuất xuất hiện nhiều tín hiệu suy giảm, làm thế nào hệ thống xác định chính xác nguyên nhân, mức độ nghiêm trọng (Severity) và độ tin cậy của bằng chứng để lựa chọn hành động bảo trì phù hợp, an toàn và tiết kiệm chi phí?''}
\end{quote}

Bốn giả thuyết khoa học (Research Hypotheses) được xây dựng phục vụ kiểm định thống kê:
\begin{itemize}
  \item \textbf{H1 (Hiệu quả chi phí và phục hồi):} Chính sách thích ứng dựa trên nhận thức suy giảm giảm thiểu có ý nghĩa thống kê số lần tái huấn luyện không cần thiết (unnecessary retraining) so với các baseline định kỳ (periodic) và dựa trên cảnh báo drift đơn lẻ (alert-driven), trong khi vẫn duy trì năng lực phục hồi hiệu năng tương đương khi gặp trôi dạt có hại.
  \item \textbf{H2 (Độ chính xác quyết định):} Khối chẩn đoán nguyên nhân đa tín hiệu (Multi-Signal Diagnosis) làm giảm tỷ lệ ra quyết định tái huấn luyện sai lầm (false retraining decisions) so với cơ chế kích hoạt chỉ dựa trên một detector duy nhất.
  \item \textbf{H3 (An toàn tự động hóa với HITL):} Cơ chế can thiệp nhận thức rủi ro kết hợp vòng lặp chuyên gia (HITL) triệt tiêu các hành vi tự động hóa mất an toàn (unsafe automation) trong các kịch bản suy giảm mơ hồ hoặc khi bằng chứng nhãn chưa đạt độ tin cậy thống kê.
  \item \textbf{H4 (Tối ưu hóa tổng thể):} Phương pháp đề xuất cải thiện vượt trội tỷ lệ đánh đổi (trade-off) giữa mức độ phục hồi hiệu năng, chi phí tính toán hạ tầng và khối lượng can thiệp thủ công so với các chính sách CT truyền thống.
\end{itemize}

\subsection{Khung phân loại nguyên nhân và Mức độ nghiêm trọng (Taxonomy \& Severity)}

Khối chẩn đoán không chỉ trả về cờ boolean $\text{Drift} \in \{\text{True}, \text{False}\}$, mà xuất ra cấu trúc chẩn đoán chính xác gồm 3 thành tố: Nguyên nhân (Cause), Mức độ nghiêm trọng (Severity), và Độ tin cậy (Confidence).

\subsubsection{Hệ thống phân loại nguyên nhân (Cause Taxonomy)}
\begin{itemize}
  \item \texttt{NORMAL / STABLE}: Hệ thống và dữ liệu ổn định, không có biến động bất thường.
  \item \texttt{BENIGN\_DRIFT}: Dữ liệu đầu vào $P(X)$ thay đổi nhưng phân phối điều kiện $P(Y|X)$ và hiệu năng thực tế không suy giảm.
  \item \texttt{HARMFUL\_DRIFT}: Dữ liệu đầu vào biến đổi kèm theo suy giảm hiệu năng thực tế có ý nghĩa thống kê trên tập nhãn xác thực.
  \item \texttt{DATA\_QUALITY\_FAILURE}: Lỗi schema, sai lệch miền giá trị, missing rate tăng vọt hoặc mất tính tươi mới dữ liệu.
  \item \texttt{CONCEPT\_SHIFT / UNEXPLAINED}: Hiệu năng thực tế suy giảm nhưng phân phối dữ liệu đầu vào $P(X)$ không đổi, đòi hỏi chuyên gia phân tích.
  \item \texttt{INSUFFICIENT\_EVIDENCE}: Tỷ lệ phủ nhãn $r_\text{labels} < r_\text{min}$ hoặc số lượng mẫu chưa đủ để khẳng định suy giảm.
  \item \texttt{MIXED / AMBIGUOUS}: Tín hiệu mâu thuẫn giữa nhiều detector hoặc đồng thời xảy ra sự cố hạ tầng và trôi dạt dữ liệu.
\end{itemize}

\subsubsection{Thang đo mức độ nghiêm trọng (Severity Levels)}
Mức độ nghiêm trọng được lượng hóa thành 5 cấp bậc $S_0$--$S_4$ dựa trên độ lớn trôi dạt ($d_\text{drift}$), mức suy giảm hiệu năng ($\Delta_\text{perf}$), khoảng tin cậy thống kê ($CI_\text{perf}$) và độ phủ nhãn:
\begin{itemize}
  \item \textbf{$S_0$ (Normal):} Hoạt động bình thường; $\Delta_\text{perf} \approx 0$.
  \item \textbf{$S_1$ (Benign / Low):} Trôi dạt đầu vào nhẹ nhưng hiệu năng ổn định, hoặc cảnh báo sớm từ phân phối confidence khi chưa có nhãn.
  \item \textbf{$S_2$ (Moderate):} Suy giảm hiệu năng từ $3\%\text{--}8\%$ kèm $d_\text{drift} > \theta_\text{drift}$ và nhãn kiểm chứng đầy đủ.
  \item \textbf{$S_3$ (Severe):} Suy giảm hiệu năng $> 8\%$ trên diện rộng hoặc vi phạm chất lượng dữ liệu nghiêm trọng.
  \item \textbf{$S_4$ (Critical):} Mô hình sụp đổ hoàn toàn hoặc vi phạm guardrails an toàn trên môi trường sản xuất.
\end{itemize}

\subsection{Thuật toán chẩn đoán và Cấu trúc dữ liệu đầu ra}\label{subsec:diagnosis}

\subsubsection{Cấu trúc dữ liệu chẩn đoán chuẩn hóa (Structured JSON Output)}
Kết quả của khối chẩn đoán được chuẩn hóa dưới dạng JSON schema có thể giải thích được và lưu vết kiểm toán:
\begin{tcolorbox}[colback=gray!4!white,colframe=gray!45!black,arc=2pt,boxrule=0.5pt,top=3pt,bottom=3pt,left=5pt,right=5pt]
\small
\begin{verbatim}
{
  "event_id": "deg_evt_8f1a2c",
  "cause": "HARMFUL_DRIFT",
  "severity": "S2_MODERATE",
  "confidence": 0.94,
  "evidence": {
    "psi": 0.32,
    "f1_delta": -0.075,
    "label_coverage": 0.86,
    "quality_passed": true
  },
  "recommended_action": "INCREMENTAL_RETRAIN",
  "action_constraints": { "cooldown_passed": true, "budget_ok": true }
}
\end{verbatim}
\end{tcolorbox}

\begin{tcolorbox}[colback=gray!3!white,colframe=gray!40!black,arc=2pt,boxrule=0.6pt,top=4pt,bottom=4pt,left=6pt,right=6pt,title=\small\textbf{Thuật toán 1: Chẩn đoán nguyên nhân suy giảm và xác định hành động bảo trì}]
\small
\textbf{Đầu vào (Input):} Thực thể EvidenceWindow $E = \{d_\text{drift}, q_\text{quality}, \Delta_\text{perf}, \Delta_\text{conf}, r_\text{labels}, n, CI_\text{perf}\}$\\[0.3em]
\textbf{Bước 1 (Kiểm định chất lượng dữ liệu - Data Quality Gate):}\\
\hspace*{1em}\textbf{If} $q_\text{quality} < \theta_\text{quality}$ \textbf{then}\\
\hspace*{2em}\textbf{Return} Output(\texttt{DATA\_QUALITY\_FAILURE}, Severity: $S_3$, Action: \texttt{DATA\_ENGINEERING}, Conf: High)\\[0.25em]
\textbf{Bước 2 (Kiểm tra độ phủ nhãn - Label Sufficiency Gate):}\\
\hspace*{1em}\textbf{If} $r_\text{labels} < r_\text{min}$ \textbf{or} $n < n_\text{min}$ \textbf{then}\\
\hspace*{2em}\textbf{If} $\Delta_\text{conf}$ lệch chuẩn có ý nghĩa \textbf{then}\\
\hspace*{3em}\textbf{Return} Output(\texttt{INSUFFICIENT\_EVIDENCE}, Severity: $S_1$, Action: \texttt{NO\_OP\_HOLD}, Conf: Low)\\
\hspace*{2em}\textbf{Else}\\
\hspace*{3em}\textbf{Return} Output(\texttt{NORMAL}, Severity: $S_0$, Action: \texttt{NO\_OP}, Conf: High)\\[0.25em]
\textbf{Bước 3 (Kiểm định trôi dạt có hại - Harmful Drift Gate):}\\
\hspace*{1em}\textbf{If} $d_\text{drift} > \theta_\text{drift}$ \textbf{and} $\Delta_\text{perf} < \theta_\text{perf}$ \textbf{and} Cận dưới $CI_\text{perf}$ khẳng định suy giảm \textbf{then}\\
\hspace*{2em}\textbf{If} $|\Delta_\text{perf}| > 0.08$ \textbf{then} Sev $\leftarrow S_3$; Act $\leftarrow$ \texttt{FULL\_RETRAIN}\\
\hspace*{2em}\textbf{Else} Sev $\leftarrow S_2$; Act $\leftarrow$ \texttt{INCREMENTAL\_RETRAIN}\\
\hspace*{2em}\textbf{If} (Cooldown OK $\wedge$ Budget OK) \textbf{then}\\
\hspace*{3em}\textbf{Return} Output(\texttt{HARMFUL\_DRIFT}, Severity: Sev, Action: Act, Conf: High)\\
\hspace*{2em}\textbf{Else}\\
\hspace*{3em}\textbf{Return} Output(\texttt{HARMFUL\_DRIFT}, Severity: Sev, Action: \texttt{HITL}, Conf: High)\\[0.25em]
\textbf{Bước 4 (Trôi dạt lành tính - Benign Drift Gate):}\\
\hspace*{1em}\textbf{If} $d_\text{drift} > \theta_\text{drift}$ \textbf{and} $|\Delta_\text{perf}| \approx 0$ \textbf{then}\\
\hspace*{2em}\textbf{Return} Output(\texttt{BENIGN\_DRIFT}, Severity: $S_1$, Action: \texttt{NO\_OP}, Conf: High)\\[0.25em]
\textbf{Bước 5 (Suy giảm hiệu năng chưa rõ nguyên nhân - Concept Shift):}\\
\hspace*{1em}\textbf{If} $\Delta_\text{perf} < \theta_\text{perf}$ \textbf{and} $d_\text{drift} \le \theta_\text{drift}$ \textbf{then}\\
\hspace*{2em}\textbf{Return} Output(\texttt{CONCEPT\_SHIFT}, Severity: $S_3$, Action: \texttt{HITL}, Conf: Medium)\\[0.25em]
\textbf{Mặc định (Fallback):}\\
\hspace*{1em}\textbf{Return} Output(\texttt{MIXED\_AMBIGUOUS}, Severity: $S_2$, Action: \texttt{HITL}, Conf: Low)
\end{tcolorbox}

\subsection{Chiến lược xử lý nhãn trễ (Delayed-Label Handling Strategy)}\label{subsec:delayed-labels}

Nhãn thực tế (ground-truth labels) thường có độ trễ lớn so với thời điểm suy luận, tạo nên điểm nghẽn nghiêm trọng nhất trong việc giám sát hiệu năng mô hình. Hệ thống đề xuất 5 nguyên tắc xử lý nhãn trễ:

\begin{enumerate}
  \item \textbf{Chuỗi liên kết mã định danh dự đoán (End-to-end Prediction ID):} Gateway \codepath{model-server} tạo mã định danh duy nhất (\codepath{prediction\_id}) và trả về cho client trong HTTP response, đồng thời gắn kèm mã này vào event stream lưu trữ. Client sử dụng chính \codepath{prediction\_id} để gửi nhãn thực tế qua Feedback API. Đây là \textbf{điều kiện tiên quyết} để thực hiện phép join giữa dự đoán và nhãn thực tế.
  \item \textbf{Cửa sổ trượt gắn mốc thời gian nhãn (Watermark-based Sliding Window):} Phép tính toán hiệu năng (F1, Precision, Recall) chỉ được thực hiện trên cửa sổ thời gian $[t-W, t]$ khi tỷ lệ phủ nhãn đạt ngưỡng tin cậy tối thiểu:
  $$r_\text{labels} = \frac{|\text{Dự đoán đã có nhãn}|}{|\text{Tổng số lượt dự đoán}|} \ge r_\text{min}$$
  Nếu $r_\text{labels} < r_\text{min}$, hệ thống ghi nhận trạng thái chưa đủ nhãn (\texttt{INSUFFICIENT\_LABELS}) và tuyệt đối không nội suy hiệu năng giả định để auto-retrain.
  \item \textbf{Chỉ số đại diện khi thiếu nhãn (Proxy Signals):} Khi nhãn chưa sẵn sàng, hệ thống dựa vào các tín hiệu đại diện: độ trôi dạt phân phối đầu ra $P(\hat{Y})$, độ dịch chuyển điểm tự tin (calibrated confidence drift) và biến động độ trễ để phát cảnh báo sớm, không dùng để kích hoạt CT trực tiếp.
  \item \textbf{Phân tầng độ tin cậy của nhãn (Label Trust Levels):} Mỗi bản ghi nhãn phản hồi đều lưu kèm nguồn thu thập (\codepath{label\_source}: người dùng cuối, chuyên gia gán nhãn, hay pipeline tự động) và mức độ tin cậy (\codepath{trust\_level}), ngăn ngừa việc trộn lẫn nhãn nhiễu vào dữ liệu huấn luyện mới.
  \item \textbf{Ước lượng thống kê với độ phủ nhãn một phần:} Khi chỉ có một phần dự đoán được gán nhãn (ví dụ 30\%), hiệu năng được tính toán kèm khoảng tin cậy 95\% ($CI_\text{perf}$). Cổng quyết định chỉ kích hoạt CT khi cận dưới của khoảng tin cậy xác nhận sự suy giảm thực sự.
\end{enumerate}
